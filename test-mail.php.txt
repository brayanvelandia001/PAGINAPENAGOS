<?php

header("Content-Type: text/plain; charset=UTF-8");

/*
|--------------------------------------------------------------------------
| CONFIGURACIÓN DE PRUEBA
|--------------------------------------------------------------------------
*/

$destinatario = "bvelandia@penagos.com";

$asunto = "PRUEBA CORREO PHP PENAGOS";

$mensaje = "
<html>
<head>
    <meta charset='UTF-8'>
</head>

<body style='font-family: Arial, sans-serif; color: #333;'>

    <h2 style='color: #2563eb;'>
        Prueba de correo PHP
    </h2>

    <hr>

    <p>
        Este correo fue enviado directamente desde PHP
        utilizando la función mail().
    </p>

    <p>
        Si recibes este correo, significa que el servidor
        puede enviar correos correctamente mediante PHP.
    </p>

    <hr>

    <p style='font-size: 12px; color: #777;'>
        Prueba realizada desde react.penagos.com
    </p>

</body>
</html>
";


/*
|--------------------------------------------------------------------------
| CABECERAS
|--------------------------------------------------------------------------
*/

$headers = [];

$headers[] = "MIME-Version: 1.0";
$headers[] = "Content-Type: text/html; charset=UTF-8";

$headers[] = "From: bvelandia@penagos.com";
$headers[] = "Reply-To: bvelandia@penagos.com";


/*
|--------------------------------------------------------------------------
| ENVIAR CORREO
|--------------------------------------------------------------------------
*/

$resultado = mail(
    $destinatario,
    $asunto,
    $mensaje,
    implode("\r\n", $headers)
);


/*
|--------------------------------------------------------------------------
| MOSTRAR RESULTADO
|--------------------------------------------------------------------------
*/

echo "========================================\n";
echo "PRUEBA DE ENVÍO DE CORREO PHP\n";
echo "========================================\n\n";

echo "Destinatario:\n";
echo $destinatario . "\n\n";

echo "Asunto:\n";
echo $asunto . "\n\n";

if ($resultado) {

    echo "RESULTADO:\n";
    echo "MAIL() DEVOLVIÓ TRUE\n\n";

    echo "El servidor PHP aceptó el envío.\n";
    echo "Ahora revisa el buzón y Track Delivery.\n";

} else {

    echo "RESULTADO:\n";
    echo "MAIL() DEVOLVIÓ FALSE\n\n";

    echo "El servidor PHP NO pudo aceptar el envío.\n";
}

echo "\n========================================\n";
